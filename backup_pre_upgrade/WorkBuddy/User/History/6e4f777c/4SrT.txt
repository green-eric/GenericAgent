def get_rating(score):
    if score >= 80: return 'A+'
    elif score >= 70: return 'A'
    elif score >= 60: return 'B+'
    elif score >= 50: return 'B'
    elif score >= 40: return 'C'
    else: return 'D'

def get_confidence(score, completeness):
    if completeness < 80: return '低（数据不足）'
    elif score >= 70: return '高'
    elif score >= 50: return '中'
    else: return '中（需验证）'

    print(f"\n评估日期: {eval_date.date()}"); print(f"股票数量: {len(symbols)}"); print(f"并发数: {workers}"); print(f"模式: {'模拟测试' if mock else '真实数据'}"); print("-" * 50)
    results = []; lock = threading.Lock(); semaphore = threading.Semaphore(workers)
    def evaluate_one(code, name_from_pool):
        semaphore.acquire()
        try:
            if mock: res = _mock_evaluate(code)
            else:
                try:
                    provider = DataProvider(); name = provider.get_stock_name(code); industry = provider.get_industry(code); df = provider.get_combined_financials(code)
                    if df.empty: return 'no_data'

                    if len(df) < 4: return 'no_data'
                    calc = IndicatorCalculator(df, eval_date=pd.Timestamp(eval_date)); quote = provider.get_stock_quote(code); scorer = Scorer(calc, quote, industry=industry); scores = scorer.total_score(); comp = calc.get_completeness_info()
                    res = {'symbol':code,'name':name,'industry':industry,'q_revenue_yoy':round(calc.q_revenue_yoy,2),'q_net_profit_yoy':round(calc.q_net_profit_yoy,2),'roe_ttm':round(calc.roe_ttm,2),'gross_margin_ttm':round(calc.gross_margin_ttm,2),'net_profit_ratio':round(calc.net_profit_ratio,2),'fcf_yield':round(calc.fcf_ttm/quote.get('total_mv',1) if quote.get('total_mv',0)>0 else 0,4),'cash_recovery_rate':round(calc.cash_recovery_rate,2),'de_ratio':round(calc.de_ratio,2),'current_ratio':round(calc.current_ratio,2),'asset_liability_ratio':round(calc.asset_liability_ratio,2),'pe_ttm':round(quote.get('pe_ttm',0),2),'total_mv':round(quote.get('total_mv',0)/1e8,2),'annual_report_date':str(df[df['report_date'].dt.month==12]['report_date'].max().date()) if len(df[df['report_date'].dt.month==12])>0 else str(df['report_date'].max().date()),'latest_quarter':str(df['report_date'].max().date()),'data_completeness':comp['score'],'quarter_coverage':comp['quarter_coverage'],'field_gaps':comp['field_gaps'],'total_score':scores['total_score'],'rating':get_rating(scores['total_score']),'confidence':get_confidence(scores['total_score'],comp['score']),'profitability':round(scores['profitability'],2),'growth':round(scores['growth'],2),'cash_flow_quality':round(scores['cash_flow'],2),'leverage_risk':round(scores['leverage'],2),'veto':'是' if scores['veto'] else '否','veto_reason':scores.get('veto_reason','')}

        finally: semaphore.release();
            if not mock: time.sleep(rate_limit)
    from concurrent.futures import ThreadPoolExecutor, as_completed
    with ThreadPoolExecutor(max_workers=workers) as executor:
        futures = {executor.submit(evaluate_one, code, name): (code, name) for code, name in symbols}
        for future in as_completed(futures):
            try: future.result()

def evaluate_from_db(symbols, eval_date, db):
    print(f"\n从数据库读取数据评分..."); print("批量预加载股票信息和行情..."); all_stock_info = {}; all_quotes = {}
    try:
        conn = sqlite3.connect(db.db_path); df_info = pd.read_sql('SELECT * FROM stocks', conn)
        for _, row in df_info.iterrows(): all_stock_info[row['symbol']] = dict(row)
        df_quotes = pd.read_sql('SELECT q1.* FROM quotes q1 INNER JOIN (SELECT symbol, MAX(trade_date) as max_date FROM quotes GROUP BY symbol) q2 ON q1.symbol=q2.symbol AND q1.trade_date=q2.max_date', conn)
        for _, row in df_quotes.iterrows(): all_quotes[row['symbol']] = {'total_mv':row.get('total_mv',0) or 0,'pe_ttm':row.get('pe_ttm',0) or 0}
        conn.close(); print(f"  已加载 {len(all_stock_info)} 条股票信息, {len(all_quotes)} 条行情")
    except Exception as e: logging.debug(f"批量预加载失败: {e}")
    try:
        from data_provider import preload_market_data; preload_market_data()
    except Exception as e: logging.debug(f"行情缓存预加载失败: {e}")
    results = []
    for i, (code, name_from_pool) in enumerate(symbols, 1):
        print(f"\r[{i}/{len(symbols)}] {code} ({name_from_pool})...", end='', flush=True)
        df = db.get_financials(code)
        if df is None or df.empty or len(df) < 4: continue
        if not pd.api.types.is_datetime64_any_dtype(df['report_date']): df['report_date'] = pd.to_datetime(df['report_date'], errors='coerce')
        if 'ann_date' in df.columns and not pd.api.types.is_datetime64_any_dtype(df['ann_date']): df['ann_date'] = pd.to_datetime(df['ann_date'], errors='coerce')
        df = df[df['ann_date'] <= pd.Timestamp(eval_date)].copy()
        if len(df) < 4: continue
        stock_info = all_stock_info.get(code, {})
        name = name_from_pool if name_from_pool and name_from_pool != code else stock_info.get('name') or code
        industry = stock_info.get('industry', None); quote = all_quotes.get(code, {'total_mv':0,'pe_ttm':0})
        need_fallback = (not name or name==code or not industry or industry=='' or quote.get('total_mv',0)==0)
        if need_fallback:
            try:
                from data_provider import DataProvider, _lookup_spot_cache, _akshare_unavailable, _info_cache
                if _akshare_unavailable:
                    if code in _info_cache: industry = _info_cache[code].get('industry') or industry
                else:
                    spot = _lookup_spot_cache(code)
                    if spot:
                        if (not name or name==code) and spot.get('name'): name = spot['name']
                        if spot.get('total_mv',0)>0: quote['total_mv'] = spot['total_mv']
                        if spot.get('pe_ttm',0)>0: quote['pe_ttm'] = spot['pe_ttm']
                    if not industry or industry=='': industry = DataProvider.get_industry(code)
                    if industry or name!=code: db.save_stock_info(code, name, industry)
                    if quote.get('total_mv',0)>0: db.save_quote(code, quote)
            except Exception as e: logging.debug(f"{code} 信息补取失败: {e}")
        calc = IndicatorCalculator(df, eval_date=pd.Timestamp(eval_date)); scorer = Scorer(calc, quote, industry=industry); scores = scorer.total_score(); comp = calc.get_completeness_info()
        res = {'symbol':code,'name':name,'industry':industry or '','q_revenue_yoy':round(calc.q_revenue_yoy,2),'q_net_profit_yoy':round(calc.q_net_profit_yoy,2),'roe_ttm':round(calc.roe_ttm,2),'gross_margin_ttm':round(calc.gross_margin_ttm,2),'net_profit_ratio':round(calc.net_profit_ratio,2),'fcf_yield':round(calc.fcf_ttm/quote.get('total_mv',1) if quote.get('total_mv',0)>0 else 0,4),'cash_recovery_rate':round(calc.cash_recovery_rate,2),'de_ratio':round(calc.de_ratio,2),'current_ratio':round(calc.current_ratio,2),'asset_liability_ratio':round(calc.asset_liability_ratio,2),'pe_ttm':round(quote.get('pe_ttm',0),2),'total_mv':round(quote.get('total_mv',0)/1e8,2),'annual_report_date':str(df[df['report_date'].dt.month==12]['report_date'].max().date()) if len(df[df['report_date'].dt.month==12])>0 else str(df['report_date'].max().date()) if hasattr(df['report_date'].max(),'date') else str(df['report_date'].max()),'latest_quarter':str(df['report_date'].max().date()) if hasattr(df['report_date'].max(),'date') else str(df['report_date'].max()),'data_completeness':comp['score'],'quarter_coverage':comp['quarter_coverage'],'field_gaps':comp['field_gaps'],'total_score':scores['total_score'],'rating':get_rating(scores['total_score']),'confidence':get_confidence(scores['total_score'],comp['score']),'profitability':round(scores['profitability'],2),'growth':round(scores['growth'],2),'cash_flow_quality':round(scores['cash_flow'],2),'leverage_risk':round(scores['leverage'],2),'veto':'是' if scores['veto'] else '否','veto_reason':scores.get('veto_reason','')}
        results.append(res)
    print(f"\n完成，共 {len(results)} 条结果"); return results

def fetch_to_db(symbols, eval_date, db, args):
    import time as _time; t_batch_start = _time.time()
    log_dir = os.path.join(os.path.dirname(os.path.abspath(__file__)), 'logs'); os.makedirs(log_dir, exist_ok=True)
    log_ts = datetime.now().strftime('%Y%m%d_%H%M%S')
    log_file = os.path.join(log_dir, f'fetch_{log_ts}.log'); fail_file = os.path.join(log_dir, f'failed_stocks_{log_ts}.txt'); report_file = os.path.join(log_dir, f'fetch_report_{log_ts}.txt')
    def log_write(msg, to_console=True):
        ts = datetime.now().strftime('%H:%M:%S'); line = f"[{ts}] {msg}"
        if to_console:
            try: print(line)
            except UnicodeEncodeError: print(line.encode('ascii', errors='replace').decode('ascii'))
        with open(log_file, 'a', encoding='utf-8') as lf: lf.write(line + '\n')
    log_write('='*60); log_write(f"数据获取启动 | workers={args.workers} rate_limit={args.rate_limit}s"); log_write('='*60)
    from data_provider import preload_market_data; preload_ok = preload_market_data(); log_write(f"[预加载] {'成功' if preload_ok else '失败'}")
    from concurrent.futures import ThreadPoolExecutor, as_completed
    symbol_codes = [code for code,_ in symbols]; full_fetch, incremental_fetch = db.get_stale_symbols(symbol_codes, min_quarters=4)
    log_write(f"[增量分析] 总池 {len(symbols)} 只 | 全量 {len(full_fetch)} | 增量 {len(incremental_fetch)}")
    fetch_symbols = full_fetch + incremental_fetch; fetch_mode = {s:'full' for s in full_fetch}; fetch_mode.update({s:'incr' for s in incremental_fetch}); code_to_name = {code:name for code,name in symbols}
    incr_existing_dates = {}
    if incremental_fetch:
        try:
            conn = sqlite3.connect(db.db_path); cur = conn.cursor(); placeholders = ','.join(['?' for _ in incremental_fetch])
            cur.execute(f'SELECT symbol, report_date FROM financials WHERE symbol IN ({placeholders})', incremental_fetch)
            for sym, rdate in cur.fetchall():
                if sym not in incr_existing_dates: incr_existing_dates[sym] = set()
                incr_existing_dates[sym].add(rdate); conn.close()
        except Exception as e: log_write(f"[预加载] 增量报告期预加载失败: {e}")
    if not fetch_symbols: log_write("所有股票数据已存在且最新，无需获取")
    else:
        success_count = fail_count = empty_count = timeout_count = network_err_count = parse_err_count = other_err_count = 0
        lock = threading.Lock(); info_batch = []; failed_symbols = []; failed_details = []
        def classify_error(e):
            ename = type(e).__name__; emsg = str(e).lower()
            if 'timeout' in ename.lower() or 'timeout' in emsg: return 'timeout'
            elif any(k in emsg for k in ['connection','refused','reset','disconnected','network','remote']): return 'network'
            elif any(k in emsg for k in ['json','parse','decode','keyerror','indexerror']): return 'parse'
            else: return 'other'
        def fetch_one(code):
            nonlocal success_count, fail_count, empty_count, timeout_count, network_err_count, parse_err_count, other_err_count
            t0 = _time.time(); mode = fetch_mode.get(code,'full'); mode_tag = '[全量]' if mode=='full' else '[增量]'; name_from_pool = code_to_name.get(code, code)
            try:
                from data_provider import DataProvider, _lookup_spot_cache, _info_cache; provider = DataProvider(); df = provider.get_combined_financials(code)
                if df is not None and not df.empty:
                    if mode == 'incr':
                        existing_dates = incr_existing_dates.get(code, set()); new_df = df[~df['report_date'].astype(str).isin(existing_dates)]
                        if new_df.empty:
                            elapsed = _time.time()-t0
                            with lock: success_count+=1; done=success_count+fail_count+empty_count; pct=done/len(fetch_symbols)*100; print(f"\r  [{done}/{len(fetch_symbols)}] {pct:.0f}% | OK:{success_count} FAIL:{fail_count} | {mode_tag} {code} 已是最新 {elapsed:.1f}s    ", end='', flush=True)
                            _time.sleep(args.rate_limit); return
                        rows_saved = db.save_financials(code, new_df)
                    else: rows_saved = db.save_financials(code, df)
                    spot = _lookup_spot_cache(code); name = name_from_pool if name_from_pool!=code else spot.get('name',code); total_mv = spot.get('total_mv',0); pe_ttm = spot.get('pe_ttm',0)
                    industry = None
                    if code in _info_cache: industry = _info_cache[code].get('industry')
                    if not industry: industry = DataProvider.get_industry(code)
                    if name!=code: db.save_stock_info(code, name, industry or '')
                    quote = {'total_mv':total_mv,'pe_ttm':pe_ttm}
                    if total_mv>0: db.save_quote(code, quote)
                    elapsed = _time.time()-t0
                    with lock: success_count+=1; info_batch.append((code,name,industry)); done=success_count+fail_count+empty_count; pct=done/len(fetch_symbols)*100; rows_info=f" {rows_saved}行" if mode=='incr' else ""; print(f"\r  [{done}/{len(fetch_symbols)}] {pct:.0f}% | OK:{success_count} FAIL:{fail_count} | {mode_tag} {code} ({name}){rows_info} {elapsed:.1f}s    ", end='', flush=True)
                else:
                    elapsed = _time.time()-t0
                    with lock: empty_count+=1; fail_count+=1; failed_symbols.append(code); failed_details.append(f"{code}\t{mode_tag}\t数据为空\t{elapsed:.1f}s"); done=success_count+fail_count+empty_count; pct=done/len(fetch_symbols)*100; print(f"\r  [{done}/{len(fetch_symbols)}] {pct:.0f}% | OK:{success_count} FAIL:{fail_count} | {mode_tag} {code} 数据为空 {elapsed:.1f}s    ", end='', flush=True)
            except Exception as e:
                elapsed = _time.time()-t0; err_type = classify_error(e)
                with lock:
                    fail_count+=1
                    if err_type=='timeout': timeout_count+=1
                    elif err_type=='network': network_err_count+=1
                    elif err_type=='parse': parse_err_count+=1
                    else: other_err_count+=1
                    failed_symbols.append(code); failed_details.append(f"{code}\t{mode_tag}\t{type(e).__name__}: {str(e)[:100]}\t{elapsed:.1f}s")
                    done=success_count+fail_count+empty_count; pct=done/len(fetch_symbols)*100
                    print(f"\r  [{done}/{len(fetch_symbols)}] {pct:.0f}% | OK:{success_count} FAIL:{fail_count} | {mode_tag} {code} [{type(e).__name__}] {elapsed:.1f}s    ", end='', flush=True)
            _time.sleep(args.rate_limit)
        log_write(f"开始并发获取 {len(fetch_symbols)} 只 (workers={args.workers})...")
        with ThreadPoolExecutor(max_workers=args.workers) as executor:
            futures = {executor.submit(fetch_one, s): s for s in fetch_symbols}
            for f in as_completed(futures):
                try: f.result()
                except Exception as e: sym=futures[f]; log_write(f"  [FATAL] {sym}: {type(e).__name__}: {e}"); lock.acquire(); fail_count+=1; failed_symbols.append(sym); lock.release()
        print()
        if info_batch: db.save_stock_info_batch(info_batch); log_write(f"股票信息批量保存完成: {len(info_batch)} 条")
        t_batch_elapsed = _time.time()-t_batch_start; avg_time = t_batch_elapsed/len(fetch_symbols) if fetch_symbols else 0
        rate_str = f"{success_count/len(fetch_symbols)*100:.1f}%" if fetch_symbols else "N/A"
        report_lines = ['','='*60,'数据获取汇总报告','='*60,f"时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}",f"总池: {len(symbols)} 只 | 全量: {len(full_fetch)} | 增量: {len(incremental_fetch)}",f"成功: {success_count} ({rate_str}) | 失败: {fail_count}",f"  - 数据为空:{empty_count} 网络:{network_err_count} 超时:{timeout_count} 解析:{parse_err_count} 其他:{other_err_count}",f"耗时: {t_batch_elapsed:.0f}s | 平均: {avg_time:.1f}s/只"]
        if failed_symbols: report_lines.append(f"失败: {', '.join(failed_symbols[:20])}{'...' if len(failed_symbols)>20 else ''}")
        for l in report_lines: log_write(l)
        with open(report_file, 'w', encoding='utf-8') as rf: rf.write('\n'.join(report_lines))
        if failed_symbols:
            with open(fail_file, 'w', encoding='utf-8') as ff:
                ff.write(f'# 失败股票列表\n');
                for s in failed_symbols: ff.write(s+'\n')
    if args.fetch_only:
        stats = db.get_db_stats(); log_write(f"[DB统计] 财务记录: {stats['financials']} | 股票数: {stats['symbols_with_fin']} | DB大小: {stats['db_size_mb']}MB")
        if failed_symbols: log_write(f"[提示] 重跑: python main.py --real --pool {fail_file} --workers 8 --rate-limit 0.1 --save-db --fetch-only --db {args.db}")
        return

def main():
    import sys as _sys; _sys.stdout.reconfigure(encoding='utf-8', errors='replace'); _sys.stderr.reconfigure(encoding='utf-8', errors='replace')
    _log_dir = os.path.join(os.path.dirname(os.path.abspath(__file__)), 'logs'); os.makedirs(_log_dir, exist_ok=True)
    _log_ts = datetime.now().strftime('%Y%m%d_%H%M%S'); _log_path = os.path.join(_log_dir, f'run_{_log_ts}.log')
    _root_logger = logging.getLogger(); _root_logger.setLevel(logging.INFO); _root_logger.handlers.clear()
    _file_handler = logging.FileHandler(_log_path, encoding='utf-8'); _file_handler.setFormatter(logging.Formatter('%(asctime)s [%(levelname)s] %(message)s', datefmt='%H:%M:%S'))
    _console_handler = logging.StreamHandler(sys.stdout); _console_handler.setFormatter(logging.Formatter('%(asctime)s [%(levelname)s] %(message)s', datefmt='%H:%M:%S'))
    _root_logger.addHandler(_file_handler); _root_logger.addHandler(_console_handler)
    logging.info(f"日志文件: {_log_path}")

    parser = argparse.ArgumentParser(description='A股智能选股系统 V6.0.0')
    parser.add_argument('--stocks', nargs='+', help='股票代码列表')
    parser.add_argument('--pool', type=str, default='stock_pool.txt', help='股票池文件')
    parser.add_argument('--workers', type=int, default=5, help='并发线程数')
    parser.add_argument('--rate-limit', type=float, default=0.3, help='请求间隔(秒)')
    parser.add_argument('--date', type=str, help='评估日期 (YYYY-MM-DD)')
    parser.add_argument('--output', type=str, help='输出Excel文件名')
    parser.add_argument('--disable-veto', action='store_true', help='禁用否决机制')
    parser.add_argument('--mock', action='store_true', help='模拟模式（无需网络）')
    parser.add_argument('--real', action='store_true', help='真实数据模式')
    parser.add_argument('--db', type=str, default='stock_data.db', help='数据库路径')
    parser.add_argument('--save-db', action='store_true', help='保存结果到数据库')
    parser.add_argument('--from-db', action='store_true', help='从数据库读取数据评分')
    parser.add_argument('--fetch-only', action='store_true', help='仅获取数据存DB，不评分')
    args = parser.parse_args()

    if args.disable_veto: VETO_RULES['enable'] = False; print("[提示] 一票否决机制已禁用")

    if args.stocks: symbols = [(s, s) for s in args.stocks]
    else:
        from utils import load_stock_pool; symbols = load_stock_pool(args.pool)
    if not symbols: print("股票池为空"); return

    if args.date: eval_date = datetime.strptime(args.date, '%Y-%m-%d')
    else: eval_date = datetime.today()

    db = None
    if args.save_db or args.from_db or args.fetch_only:
        from database import StockDatabase; db = StockDatabase(args.db); print(f"[数据库] {args.db}")

    if args.from_db:
        results = evaluate_from_db(symbols, eval_date, db)
        if results: save_to_excel(results, args.output or None)
        return

    if args.real and db:
        fetch_to_db(symbols, eval_date, db, args)
        if args.fetch_only: return

    if args.real and not db:
        results = batch_evaluate(symbols, eval_date, mock=False, db=db, workers=args.workers, rate_limit=args.rate_limit)
    else:
        results = batch_evaluate(symbols, eval_date, mock=True, db=db, workers=args.workers, rate_limit=args.rate_limit)

    if results: save_to_excel(results, args.output or None)
    else: print("无结果可保存")

if __name__ == '__main__':
    main()
